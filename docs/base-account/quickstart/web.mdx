import { useEffect, useState } from "react";
import { sdk } from "@farcaster/miniapp-sdk";

function App() {
  const [user, setUser] = useState(null);

  useEffect(() => {
    // Uygulama yüklendiğinde Farcaster'a hazır olduğunu bildir
    sdk.actions.ready();
  }, []);

  const handleSignIn = async () => {
    try {
      const result = await sdk.actions.signIn();
      setUser(result);
      console.log("Kullanıcı oturum açtı:", result);
    } catch (err) {
      console.error("Sign in hatası:", err);
    }
  };

  return (
    <div style={{ textAlign: "center", padding: "40px" }}>
      <h1>Merhaba Farcaster 👋</h1>

      {!user ? (
        <button onClick={handleSignIn}>Farcaster ile Giriş Yap</button>
      ) : (
        <div>
          <p>Hoş geldin, {user.username || "Anonim"}!</p>
        </div>
      )}
    </div>
  );
}

export default App;
